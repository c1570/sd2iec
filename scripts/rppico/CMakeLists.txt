# sd2iec - Pico SDK glue for the rppico architecture
#
# Make compiles all sd2iec sources into libsd2iec.a. This project
# lets the Pico SDK build its own libraries, exports the SDK compile
# flags to Make (SD2IEC_FLAGS_MK) and links the final image, so the
# SDK keeps control of startup code, linker script, boot block and
# UF2 conversion.
#
# Required cache variables (set by scripts/rppico/targets.mk):
#   PICO_SDK_PATH, PICO_PLATFORM, PICO_BOARD,
#   SD2IEC_LIB, SD2IEC_FLAGS_MK

cmake_minimum_required(VERSION 3.13)

set(SD2IEC_SDK_VERSION 2.3.1)

include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(sd2iec C CXX ASM)

pico_sdk_init()

if (NOT PICO_SDK_VERSION_STRING VERSION_EQUAL SD2IEC_SDK_VERSION)
  message(WARNING "sd2iec is tested with Pico SDK ${SD2IEC_SDK_VERSION}, "
                  "found ${PICO_SDK_VERSION_STRING}")
endif()

# add_executable needs at least one source file
set(SD2IEC_STUB ${CMAKE_CURRENT_BINARY_DIR}/sd2iec-stub.c)
file(WRITE ${SD2IEC_STUB} "/* sd2iec code comes from libsd2iec.a */\n")

add_executable(sd2iec ${SD2IEC_STUB})

target_link_libraries(sd2iec
  -Wl,--whole-archive ${SD2IEC_LIB} -Wl,--no-whole-archive
  pico_runtime
  hardware_clocks
  hardware_exception
  hardware_flash
  hardware_gpio
  hardware_i2c
  hardware_irq
  hardware_spi
  hardware_sync
  hardware_timer
  hardware_uart
  hardware_watchdog
)

set_target_properties(sd2iec PROPERTIES LINK_DEPENDS ${SD2IEC_LIB})

# The EEPROM emulation addresses its data as offsets from 0
target_link_options(sd2iec PRIVATE
  -Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/eeprom.ld
  -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/sd2iec.map,--cref
)

# Run the whole image from RAM: no flash cache timing jitter in the
# bus code, and code keeps running while flash is erased
pico_set_binary_type(sd2iec copy_to_ram)

pico_add_extra_outputs(sd2iec)

# Export the compile flags of the sd2iec target for Make. SDK include
# directories are passed as system directories so the strict sd2iec
# warning flags do not apply to SDK headers.
string(REPLACE " " ";" SD2IEC_ARCH_FLAGS "${CMAKE_C_FLAGS}")
set(SD2IEC_INCLUDES "$<TARGET_PROPERTY:sd2iec,INCLUDE_DIRECTORIES>")
set(SD2IEC_DEFINES  "$<TARGET_PROPERTY:sd2iec,COMPILE_DEFINITIONS>")
file(GENERATE OUTPUT ${SD2IEC_FLAGS_MK} CONTENT
"# generated by scripts/rppico/CMakeLists.txt, do not edit
PICO_ARCH_FLAGS := ${CMAKE_C_FLAGS}
PICO_INCLUDES := $<$<BOOL:${SD2IEC_INCLUDES}>:-isystem $<JOIN:${SD2IEC_INCLUDES}, -isystem >>
PICO_DEFINES := $<$<BOOL:${SD2IEC_DEFINES}>:'-D$<JOIN:${SD2IEC_DEFINES},' '-D>'>
")
